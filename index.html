Easy Integration

Plug-and-play setup. Copy, paste, and run your command flawlessly.

Step 1

Acquire Secret Key

Open a ticket inside Discord to grab your GenKey, then execute /setup to claim your authorized Secret Token.

Step 2

Enable Studio Network

Navigate to Roblox Studio Game Settings > Security, and toggle Allow HTTP Requests to True.

Step 3

Deploy Handler

Create a ServerScript in ServerScriptService and paste the production code seamlessly.

ServerScriptService/RankHandler.lua 
--!strict
local HttpService = game:GetService("HttpService")
local Players = game:GetService("Players")

-- --- [A] Parameter Settings Configuration ---
local API_URL = "https://rankflow-r1id.onrender.com/api/v2/rank/update" 
local WEBHOOK_URL = "YOUR_DISCORD_WEBHOOK_HERE"
local SECRET_TOKEN = "OUR_SECRET_TOKEN_HERE" 

local ADMIN_ID =  -- Alternative administrative fall-back User ID
local GROUP_ID =    -- Your Roblox Group ID
local MIN_RANK =           -- Minimum rank numeric threshold required for command execution

local cooldowns = {}

-- --- [B] Webhook Log Transmission ---
local function sendWebhook(adminPlayer: Player, targetName: string, action: string)
    local data = {
        ["embeds"] = {{
            ["title"] = "🛠️ Rank Management Log",
            ["fields"] = {
                {["name"] = "Admin", ["value"] = adminPlayer.Name, ["inline"] = true},
                {["name"] = "Target", ["value"] = targetName, ["inline"] = true},
                {["name"] = "Action", ["value"] = action:upper(), ["inline"] = true}
            },
            ["color"] = action == "promote" and 65280 or 16711680,
            ["footer"] = {["text"] = "RankFlow System • " .. os.date("%X")}
        }}
    }
    pcall(function()
        HttpService:PostAsync(WEBHOOK_URL, HttpService:JSONEncode(data))
    end)
end

-- --- [C] Invoke Render Endpoint API ---
local function callRankApi(targetId: number, action: string)
    local data = { userId = targetId, action = action }
    local success, response = pcall(function()
        return HttpService:RequestAsync({
            Url = API_URL,
            Method = "POST",
            Headers = {
                ["Content-Type"] = "application/json",
                ["Authorization"] = SECRET_TOKEN
            },
            Body = HttpService:JSONEncode(data)
        })
    end)
    if success and response and response.StatusCode == 200 then
        return true
    else
        warn("❌ Rank API Error: " .. (response and response.Body or "No response received"))
        return false
    end
end

-- --- [D] Player Message Chat Listener ---
Players.PlayerAdded:Connect(function(player)
    player.Chatted:Connect(function(msg)
        local hasPermission = false
        pcall(function()
            if player:GetRankInGroup(GROUP_ID) >= MIN_RANK or player.UserId == ADMIN_ID then
                hasPermission = true
            end
        end)
        if not hasPermission then return end
        if cooldowns[player.UserId] and tick() - cooldowns[player.UserId] < 5 then return end

        local args = string.split(msg, " ")
        local cmd = args[1]:lower()
        local targetName = args[2]

        if (cmd == "!promote" or cmd == "!demote") and targetName then
            cooldowns[player.UserId] = tick()
            local cleanCmd = cmd:sub(2)

            local getSuccess, targetUserId = pcall(function()
                return Players:GetUserIdFromNameAsync(targetName)
            end)

            if getSuccess and targetUserId then
                local apiSuccess = callRankApi(targetUserId, cleanCmd)
                if apiSuccess then
                    sendWebhook(player, targetName, cleanCmd)
                    print("Successfully updated rank structure for " .. targetName)
                end
            else
                warn("❌ User non-existent: " .. tostring(targetName))
            end
        end
    end)
end)

API Reference

Direct HTTPS access endpoints designed for advanced engineering requirements.

POST /api/v2/rank/update

Parameter Type Required Description
userId Integer YES The numerical Roblox User Identification Code to manage.
action String YES Target operation. Must define strictly as "promote" or "demote".
Authorization Header YES Bearer authentication scheme payload (Bearer YOUR_TOKEN).
